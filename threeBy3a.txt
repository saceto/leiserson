Fixed Matrix Code
Here’s the corrected calculateMatrices function with the matrix multiplication bug fixed:

javascript

Copy
function calculateMatrices() {
    // Get user input for Matrix A
    const a11 = parseInt(document.getElementById('a11').value);
    const a12 = parseInt(document.getElementById('a12').value);
    const a21 = parseInt(document.getElementById('a21').value);
    const a22 = parseInt(document.getElementById('a22').value);

    if (isNaN(a11) || isNaN(a12) || isNaN(a21) || isNaN(a22)) {
        alert("Please enter valid numbers for Matrix A.");
        return;
    }

    // Initialize matrices
    let matrixA = [[a11, a12], [a21, a22]];
    let matrixB = [[0, 0], [0, 0]];
    let matrixC = [[0, 0], [0, 0]];
    let matrixSum = [[0, 0], [0, 0]];
    let matrixResult = [[0, 0], [0, 0]];

    // Generate random values for Matrix B and C (-10 to 10)
    for (let i = 0; i < 2; i++) {
        for (let j = 0; j < 2; j++) {
            matrixB[i][j] = Math.floor(Math.random() * 21) - 10;
            matrixC[i][j] = Math.floor(Math.random() * 21) - 10;
        }
    }

    // Add Matrix A and B
    for (let i = 0; i < 2; i++) {
        for (let j = 0; j < 2; j++) {
            matrixSum[i][j] = matrixA[i][j] + matrixB[i][j];
        }
    }

    // Multiply (A + B) by C
    for (let i = 0; i < 2; i++) {
        for (let j = 0; j < 2; j++) {
            matrixResult[i][j] = 0; // Reset to avoid accumulation
            for (let k = 0; k < 2; k++) {
                matrixResult[i][j] += matrixSum[i][k] * matrixC[k][j];
            }
        }
    }

    // Display matrices
    displayMatrix('matrixA', matrixA);
    displayMatrix('matrixB', matrixB);
    displayMatrix('matrixSum', matrixSum);
    displayMatrix('matrixC', matrixC);
    displayMatrix('matrixResult', matrixResult);

    document.getElementById('result').style.display = 'block';
}
